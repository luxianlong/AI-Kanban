三聚合

classDiagram
  class Dossier {
    聚合根 · 唯一的
    +Path path
    +raw() images() facts_dir()
    +facts 内存快照
    +forbidden() 纯过滤
  }
  class Fact {
    值对象 frozen · 11 字段
    +claim +grade +verbatim
    +cui_bono +why_they_did_it
    +from_json() 认不出的键会吵
  }
  class FactGrade {
    枚举 · 红线定义处
    实 实·单方 网 传 假
    +writable() 只拦【假】
  }
  class DossierRepository {
    端口 ABC
    +load() ensure()
    +raw_files() read_facts()
  }
  class EventStore {
    端口 ABC
    +append() read() last()
  }
  class FileSystemDossierRepository {
    实现 · infrastructure/
    0 个命令行入口调用
  }
  class JsonlEventStore {
    实现 · infrastructure/
    publish_page 在用
  }
  class SourceAdapter {
    端口 ABC · 防腐层
    +emit_js() normalize()
  }
  class Item {
    值对象 · 适配器产出
  }
  class Outline {
    值对象 frozen · delivery/domain
  }
  class Candidate {
    值对象 · discovery/domain
    六条拒绝规则
  }
  class EditorialLoop {
    编辑来回 · 没有类
    规则在 prompts/ 三份 md
    editorial.js 只管循环
  }

  Dossier "1" *-- "0..*" Fact
  Fact --> FactGrade
  DossierRepository ..> Dossier
  EventStore ..> Dossier
  FileSystemDossierRepository ..|> DossierRepository
  JsonlEventStore ..|> EventStore
  SourceAdapter ..> Item
  Candidate ..> Item
